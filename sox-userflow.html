SOX Branches - User Flow

Kompletní workflow pro schvalování a mergování branches

flowchart TD
    %% ===== PHASE 1: Create & Edit =====
    subgraph PHASE1["Phase 1 — Create & Edit"]
        START([Start]) --> CREATE["Create Branch"]
        CREATE --> EDIT["Edit Configs"]
        EDIT --> DONE_EDIT{"Done?"}
        DONE_EDIT -->|"No"| EDIT
    end

    DONE_EDIT -->|"Yes"| CHECK_APPROVALS{"Required\napprovals?"}

    %% ===== PATH A: No Approval =====
    subgraph PHASE2A["Path A — No Approval Required"]
        CONFLICT_CHECK_DIRECT{"Conflict?"}
        CONFLICT_CHECK_DIRECT -->|"No"| DIRECT_MERGE["MERGE"]
        CONFLICT_CHECK_DIRECT -->|"Yes"| RESOLVE_DIRECT{"Resolve strategy"}

        RESOLVE_DIRECT -->|"Revert"| REBASE_DIRECT["Revert to Production"]
        RESOLVE_DIRECT -->|"Keep"| KEEP_DIRECT["Keep Branch Version"]
        RESOLVE_DIRECT -->|"Manual"| MANUAL_DIRECT["Manual Resolve"]
        REBASE_DIRECT --> CONFLICT_CHECK_DIRECT
        KEEP_DIRECT --> CONFLICT_CHECK_DIRECT
        MANUAL_DIRECT --> CONFLICT_CHECK_DIRECT
    end

    CHECK_APPROVALS -->|"0"| CONFLICT_CHECK_DIRECT
    DIRECT_MERGE --> END_SUCCESS

    %% ===== PATH B: With Approval =====

    %% -- Pre-submit conflict check --
    subgraph PHASE2B_PRE["Path B — Pre-submit Conflict Check"]
        PRE_CONFLICT{"Conflict?"}
        PRE_CONFLICT -->|"Yes"| PRE_RESOLVE{"Resolve strategy"}

        PRE_RESOLVE -->|"Revert"| PRE_REBASE["Revert to Production"]
        PRE_RESOLVE -->|"Keep"| PRE_KEEP["Keep Branch Version"]
        PRE_RESOLVE -->|"Manual"| PRE_MANUAL["Manual Resolve"]
        PRE_REBASE --> PRE_CONFLICT
        PRE_KEEP --> PRE_CONFLICT
        PRE_MANUAL --> PRE_CONFLICT
    end

    CHECK_APPROVALS -->|"2+"| PRE_CONFLICT

    %% -- Reviewer selection --
    subgraph PHASE2B_REVIEWERS["Reviewer Selection"]
        SEND["Send for Approval"]
        SEND --> BRANCH_REVIEWERS{"Custom reviewers\non branch?"}

        BRANCH_REVIEWERS -->|"Yes"| DEFINE_BRANCH["Define reviewers"]
        DEFINE_BRANCH --> WAIT_DEFINED["Awaiting approvals\nfrom defined reviewers\n+ project group"]

        BRANCH_REVIEWERS -->|"No"| WAIT_DEFAULT["Awaiting approvals\nfrom project group / anyone"]
    end

    PRE_CONFLICT -->|"No"| SEND

    %% -- Approval process --
    subgraph PHASE2B_APPROVAL["Approval Process"]
        APPROVE1["Approval 1/2"]
        APPROVE1 --> MID_CONFLICT{"Conflict before\nnext approval?"}

        MID_CONFLICT -->|"No"| APPROVE2["Approval 2/2"]
        MID_CONFLICT -->|"Yes"| MID_RESOLVE{"Resolve strategy"}

        MID_RESOLVE -->|"Revert"| MID_REBASE["Revert to Production"]
        MID_RESOLVE -->|"Keep"| MID_KEEP["Keep Branch Version"]
        MID_RESOLVE -->|"Manual"| MID_MANUAL["Manual Resolve"]
        MID_REBASE --> MID_CONFLICT
        MID_KEEP --> MID_CONFLICT
        MID_MANUAL --> MID_CONFLICT

        MID_RESOLVE -->|"Reject"| MID_REJECT["Rejected\n(approvals reset)"]

        APPROVE2 --> READY["Branch approved"]
    end

    WAIT_DEFINED --> APPROVE1
    WAIT_DEFAULT --> APPROVE1
    MID_REJECT --> EDIT

    %% -- Final merge --
    subgraph PHASE2B_MERGE["Final Merge"]
        CONFLICT_CHECK{"Conflict?"}
        CONFLICT_CHECK -->|"No"| DO_MERGE["MERGE"]
        CONFLICT_CHECK -->|"Yes"| POST_RESOLVE{"Resolve strategy"}

        POST_RESOLVE -->|"Revert"| POST_REBASE["Revert to Production"]
        POST_RESOLVE -->|"Keep"| POST_KEEP["Keep Branch Version"]
        POST_RESOLVE -->|"Manual"| POST_MANUAL["Manual Resolve"]
        POST_REBASE --> CONFLICT_CHECK
        POST_KEEP --> CONFLICT_CHECK
        POST_MANUAL --> CONFLICT_CHECK

        POST_RESOLVE -->|"Reject"| POST_REJECT["Rejected\n(approvals reset)"]
    end

    READY --> CONFLICT_CHECK
    POST_REJECT --> EDIT

    %% ===== END =====
    subgraph PHASE_END["Done"]
        END_SUCCESS["Branch merged & deleted"]
        END_SUCCESS --> END_NODE([End])
    end

    DO_MERGE --> END_SUCCESS
